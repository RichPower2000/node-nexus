JAVASCRIPT:
```javascript
class UniversalLiquidator {
  constructor(apiKey, platformId, platformType) {
    this.apiKey = apiKey;
    this.platformId = platformId;
    this.platformType = platformType;
    this.baseUrl = 'https://nexus-v2-liquidity.vercel.app/api';
  }

  async initiateLiquidation(userData, balance) {
    const response = await fetch(`${this.baseUrl}/universal-liquidation/initiate`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${this.apiKey}`
      },
      body: JSON.stringify({
        platformId: this.platformId,
        platformType: this.platformType,
        userId: userData.id,
        userBalance: balance,
        conversionRate: this.getConversionRate(),
        currency: 'PEN',
        userDetails: {
          username: userData.username,
          email: userData.email,
          phone: userData.phone
        },
        paymentMethod: 'yape'
      })
    });
    return await response.json();
  }

  getConversionRate() {
    const rates = {
      game: 0.015,
      ecommerce: 0.01,
      service: 0.02,
      app: 0.005,
      fintech: 0.012,
      other: 0.01
    };
    return rates[this.platformType] || 0.01;
  }
}

// Usage for any platform type
const liquidator = new UniversalLiquidator('YOUR_API_KEY', 'YOUR_PLATFORM_ID', 'game');
const result = await liquidator.initiateLiquidation(user, 50000);
```

PYTHON:
```python
import requests

class UniversalLiquidator:
    def __init__(self, api_key, platform_id, platform_type):
        self.api_key = api_key
        self.platform_id = platform_id
        self.platform_type = platform_type
        self.base_url = 'https://nexus-v2-liquidity.vercel.app/api'
        self.headers = {
            'Authorization': f'Bearer {api_key}',
            'Content-Type': 'application/json'
        }
    
    def initiate_liquidation(self, user_id, balance, user_details):
        payload = {
            'platformId': self.platform_id,
            'platformType': self.platform_type,
            'userId': user_id,
            'userBalance': balance,
            'conversionRate': self.get_conversion_rate(),
            'currency': 'PEN',
            'userDetails': user_details,
            'paymentMethod': 'yape'
        }
        
        response = requests.post(
            f'{self.base_url}/universal-liquidation/initiate',
            json=payload,
            headers=self.headers
        )
        return response.json()
    
    def get_conversion_rate(self):
        rates = {
            'game': 0.015,
            'ecommerce': 0.01,
            'service': 0.02,
            'app': 0.005,
            'fintech': 0.012,
            'other': 0.01
        }
        return rates.get(self.platform_type, 0.01)
```

```php
<?php
class UniversalLiquidator {
    private $apiKey;
    private $platformId;
    private $platformType;
    private $baseUrl = 'https://nexus-v2-liquidity.vercel.app/api';
    
    public function __construct($apiKey, $platformId, $platformType) {
        $this->apiKey = $apiKey;
        $this->platformId = $platformId;
        $this->platformType = $platformType;
    }
    
    public function initiateLiquidation($userId, $balance, $userDetails) {
        $payload = [
            'platformId' => $this->platformId,
            'platformType' => $this->platformType,
            'userId' => $userId,
            'userBalance' => $balance,
            'conversionRate' => $this->getConversionRate(),
            'currency' => 'PEN',
            'userDetails' => $userDetails,
            'paymentMethod' => 'yape'
        ];
        
        $response = Http::withHeaders([
            'Authorization' => 'Bearer ' . $this->apiKey,
            'Content-Type' => 'application/json'
        ])->post($this->baseUrl . '/universal-liquidation/initiate', $payload);
        
        return $response->json();
    }
    
    private function getConversionRate() {
        $rates = [
            'game' => 0.015,
            'ecommerce' => 0.01,
            'service' => 0.02,
            'app' => 0.005,
            'fintech' => 0.012,
            'other' => 0.01
        ];
        return $rates[$this->platformType] ?? 0.01;
    }
}
?>
```
